```ts
/** 删除商品：必须已下架、无未完成订单、所有 SKU 库存为 0 */
export function deleteProduct(db: DB, id: number, session: AdminSession): Product {
  const product = db.product.find((p) => p.id === id && !p.deleted)
  if (!product) throw notFound('商品不存在')
  if (product.status === 1) throw bad('请先下架商品再删除')
  assertNoOpenOrders(db, { productId: id })

  const skus = db.productSku.filter((s) => !s.deleted && s.productId === id)
  const withStock = skus.filter((s) => s.stockWeight > 0)
  if (withStock.length > 0) {
    throw bad(`该商品下 ${withStock.length} 个 SKU 仍有库存，请先出库或盘点为 0`)
  }

  product.deleted = true
  product.updatedAt = new Date().toISOString()
  product.updatedBy = session.username
  for (const sku of skus) {
    sku.deleted = true
    sku.updatedAt = product.updatedAt
  }
  return product
}

/** 上架 / 下架；上架前校验主图和启用 SKU */
export function setProductStatus(db: DB, id: number, status: 0 | 1, session: AdminSession): Product {
  const product = db.product.find((p) => p.id === id && !p.deleted)
  if (!product) throw notFound('商品不存在')
  if (status === 1) {
    if (!product.cover) throw bad('上架前请先上传主图')
    const hasActiveSku = db.productSku.some(
      (s) => !s.deleted && s.productId === id && s.status === 1,
    )
    if (!hasActiveSku) throw bad('上架前请至少启用一个 SKU')
  }
  product.status = status
  product.updatedAt = new Date().toISOString()
  product.updatedBy = session.username
  return product
}
```

```ts
import { addLog, handle, ok, readJson, requireAdmin, requireId, toStatus } from '@/lib/api'
import { mutate } from '@/lib/store'

  return handle(async () => {
    const session = requireAdmin('product')
    const id = requireId(params.id)
    const body = await readJson<{ status?: unknown }>(req)

        module: '商品', action: status === 1 ? '上架' : '下架',
        requestUri: `/api/admin/products/${id}/status`,
      })
    })
    return ok(true, status === 1 ? '已上架' : '已下架')
  })
```

```ts
// 商品与库存管理的查询：PC 后台与小程序管理端共用
import { notFound } from './api'
import { sumWeight } from './num'
import { availableWeight, productStockLevel, stockLevelOf } from './stock'
import type { DB, Product } from './types'

export interface ProductFilter {
  keyword?: string
  categoryId?: number
  materialId?: number
  status?: 0 | 1
}

export function productFilterFromUrl(url: URL): ProductFilter {
  const statusParam = url.searchParams.get('status')
  return {
    keyword: url.searchParams.get('keyword') ?? '',
    categoryId: Number(url.searchParams.get('categoryId') ?? 0) || 0,
    materialId: Number(url.searchParams.get('materialId') ?? 0) || 0,
    status: statusParam === '0' || statusParam === '1' ? (Number(statusParam) as 0 | 1) : undefined,
  }
}

export function filterProducts(db: DB, filter: ProductFilter): Product[] {
  const keyword = (filter.keyword ?? '').trim().toLowerCase()
  let list = db.product.filter((p) => !p.deleted)
  if (keyword) {
    list = list.filter(
      (p) => p.name.toLowerCase().includes(keyword) || p.styleNo.toLowerCase().includes(keyword),
    )
  }
  if (filter.categoryId) {
    const scope = [
      filter.categoryId,
      ...db.category.filter((c) => c.parentId === filter.categoryId).map((c) => c.id),
    ]
    list = list.filter((p) => scope.includes(p.categoryId))
  }
  if (filter.materialId) list = list.filter((p) => p.materialId === filter.materialId)
  if (filter.status !== undefined) list = list.filter((p) => p.status === filter.status)
  return [...list].sort((a, b) => a.sort - b.sort || b.id - a.id)
}

export function productRow(db: DB, p: Product) {
  const skus = db.productSku.filter((s) => !s.deleted && s.productId === p.id)
  return {
    id: p.id,
    name: p.name,
    styleNo: p.styleNo,
    cover: p.cover,
    categoryName: db.category.find((c) => c.id === p.categoryId)?.name ?? '',
    materialName: db.goldMaterial.find((m) => m.id === p.materialId)?.name ?? '',
    isRecommend: p.isRecommend,
    status: p.status,
    sort: p.sort,
    skuCount: skus.length,
    stockWeight: sumWeight(skus.map((s) => s.stockWeight)),
    availableWeight: sumWeight(skus.map((s) => availableWeight(s))),
    stockLevel: productStockLevel(skus),
    updatedAt: p.updatedAt,
  }
}

export function productDetail(db: DB, id: number) {
  const product = db.product.find((p) => p.id === id && !p.deleted)
  if (!product) throw notFound('商品不存在')

  const skus = db.productSku
    .filter((s) => !s.deleted && s.productId === id)
    .sort((a, b) => a.id - b.id)
    .map((s) => ({
      id: s.id,
      skuCode: s.skuCode,
      specName: s.specName,
      refWeight: s.refWeight,
      refQty: s.refQty,
      stockWeight: s.stockWeight,
      lockedWeight: s.lockedWeight,
      availableWeight: availableWeight(s),
      warnWeight: s.warnWeight,
      status: s.status,
      level: stockLevelOf(s),
    }))

  return { ...product, skus }
}

/** 商品表单的选择器数据，含已停用项，编辑老商品时才能显示原值 */
export function productOptions(db: DB) {
  const all = db.category.filter((c) => !c.deleted)
  const bySort = <T extends { sort: number; id: number }>(a: T, b: T) => a.sort - b.sort || a.id - b.id
  const categories: Array<{ id: number; label: string; status: 0 | 1 }> = []
  for (const parent of all.filter((c) => c.parentId === 0).sort(bySort)) {
    categories.push({ id: parent.id, label: parent.name, status: parent.status })
    for (const child of all.filter((c) => c.parentId === parent.id).sort(bySort)) {
      categories.push({ id: child.id, label: `${parent.name} / ${child.name}`, status: child.status })
    }
  }
  const materials = db.goldMaterial
    .filter((m) => !m.deleted)
    .sort(bySort)
    .map((m) => ({ id: m.id, name: m.name, purity: m.purity, status: m.status }))
  return { categories, materials }
}

export interface StockFilter {
  keyword?: string
  materialId?: number
  onlyWarn?: boolean
}

export function stockFilterFromUrl(url: URL): StockFilter {
  return {
    keyword: url.searchParams.get('keyword') ?? '',
    materialId: Number(url.searchParams.get('materialId') ?? 0) || 0,
    onlyWarn: url.searchParams.get('onlyWarn') === '1',
  }
}

export function stockRows(db: DB, filter: StockFilter) {
  const keyword = (filter.keyword ?? '').trim().toLowerCase()
  let rows = db.productSku
    .filter((s) => !s.deleted)
    .map((s) => {
      const product = db.product.find((p) => p.id === s.productId)
      const material = db.goldMaterial.find((m) => m.id === product?.materialId)
      return {
        id: s.id,
        productId: s.productId,
        productName: product?.name ?? '',
        styleNo: product?.styleNo ?? '',
        cover: product?.cover ?? '',
        productStatus: product?.status ?? 0,
        materialId: product?.materialId ?? 0,
        materialName: material?.name ?? '',
        skuCode: s.skuCode,
        specName: s.specName,
        refWeight: s.refWeight,
        refQty: s.refQty,
        stockWeight: s.stockWeight,
        lockedWeight: s.lockedWeight,
        availableWeight: availableWeight(s),
        warnWeight: s.warnWeight,
        status: s.status,
        level: stockLevelOf(s),
        updatedAt: s.updatedAt,
      }
    })

  if (keyword) {
    rows = rows.filter(
      (r) =>
        r.productName.toLowerCase().includes(keyword) ||
        r.styleNo.toLowerCase().includes(keyword) ||
        r.skuCode.toLowerCase().includes(keyword) ||
        r.specName.toLowerCase().includes(keyword),
    )
  }
  if (filter.materialId) rows = rows.filter((r) => r.materialId === filter.materialId)
  if (filter.onlyWarn) rows = rows.filter((r) => r.level !== 'ENOUGH')

  rows.sort((a, b) => a.productId - b.productId || a.id - b.id)
  return rows
}

export function stockSummary(rows: ReturnType<typeof stockRows>) {
  return {
    skuCount: rows.length,
    stockWeight: sumWeight(rows.map((r) => r.stockWeight)),
    lockedWeight: sumWeight(rows.map((r) => r.lockedWeight)),
    availableWeight: sumWeight(rows.map((r) => r.availableWeight)),
    warnCount: rows.filter((r) => r.level !== 'ENOUGH').length,
  }
}
```

```ts
// 入库 / 出库 / 盘点：PC 后台与小程序管理端共用的校验和执行
import { bad, requireId, requireString } from './api'
import { parseWeight } from './num'
import { applyStockChange } from './stock'
import type { DB, StockLog, StockLogType } from './types'

export type StockOpType = Extract<StockLogType, 'IN' | 'OUT' | 'ADJUST'>

export interface StockOpBody {
  skuId?: unknown
  weight?: unknown
  reason?: string
  refNo?: string
}

export interface StockOpInput {
  skuId: number
  weight: number
  reason: string
  refNo?: string
}

export const STOCK_OP_LABEL: Record<StockOpType, string> = { IN: '入库', OUT: '出库', ADJUST: '盘点调整' }

export function parseStockOp(type: StockOpType, body: StockOpBody): StockOpInput {
  const skuId = requireId(body.skuId, 'SKU')
  const reason = requireString(body.reason, '原因', 200)
  const weight = parseWeight(body.weight)
  if (weight == null) throw bad('请填写合法的克重')
  if (type !== 'ADJUST' && weight <= 0) throw bad('克重必须大于 0')
  return { skuId, weight, reason, refNo: body.refNo ? String(body.refNo).slice(0, 50) : undefined }
}

export function runStockOp(db: DB, type: StockOpType, input: StockOpInput, operator: string): StockLog {
  return applyStockChange(db, {
    skuId: input.skuId,
    type,
    ...(type === 'ADJUST'
      ? { targetWeight: input.weight }
      : { changeWeight: type === 'IN' ? input.weight : -input.weight }),
    reason: input.reason,
    refNo: input.refNo,
    operator,
  })
}
```

```ts
/** 按文件头判断真实图片类型；小程序上传的 Content-Type 不可靠 */
export function sniffImageMime(buffer: Buffer): 'image/jpeg' | 'image/png' | 'image/webp' | null {
  if (buffer.length >= 3 && buffer[0] === 0xff && buffer[1] === 0xd8 && buffer[2] === 0xff) {
    return 'image/jpeg'
  }
  if (buffer.length >= 8 && buffer.subarray(0, 8).equals(Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]))) {
    return 'image/png'
  }
  if (
    buffer.length >= 12 &&
    buffer.toString('latin1', 0, 4) === 'RIFF' &&
    buffer.toString('latin1', 8, 12) === 'WEBP'
  ) {
    return 'image/webp'
  }
  return null
}
```

```ts
// POST /api/mall/manage/upload — 小程序上传商品图（multipart，字段 file）
import { bad, handle, ok } from '@/lib/api'
import { MAX_FILE_BYTES, saveFile, sniffImageMime } from '@/lib/files'
import { MANAGE_ROLES, requireMallUser } from '@/lib/mall-auth'
import { loadDB } from '@/lib/store'

export const dynamic = 'force-dynamic'

export async function POST(req: Request) {
  return handle(async () => {
    requireMallUser(await loadDB(), MANAGE_ROLES)

    let form: FormData
    try {
      form = await req.formData()
    } catch {
      throw bad('请使用 multipart/form-data 上传')
    }

    const entry = form.get('file')
    if (!(entry instanceof File)) throw bad('请选择要上传的图片')
    if (entry.size === 0) throw bad('图片内容为空')
    if (entry.size > MAX_FILE_BYTES) {
      throw bad(`图片不能超过 ${Math.round(MAX_FILE_BYTES / 1024)} KB`)
    }

    const buffer = Buffer.from(await entry.arrayBuffer())
    const mime = sniffImageMime(buffer)
    if (!mime) throw bad('只支持 JPG / PNG / WEBP 图片')

    const file = await saveFile({ mime, buffer, name: entry.name })
    return ok({ url: `/api/files/${file.id}`, size: file.size, mime: file.mime }, '上传成功')
  })
}
```

```js
const LEVEL = {
  ENOUGH: { text: '充足', cls: 'badge-enough' },
  LOW: { text: '偏低', cls: 'badge-low' },
  SOLD_OUT: { text: '售罄', cls: 'badge-sold-out' },
};

function levelOf(level) {
  return LEVEL[level] || LEVEL.ENOUGH;
}

function pad2(n) {
  return n < 10 ? '0' + n : String(n);
}

/** 编码留空的规格按「款号-NN」补齐，序号避开表单内已有编码 */
function fillSkuCodes(skus, styleNo) {
  const used = {};
  skus.forEach(function (s) {
    if (s.skuCode) used[s.skuCode] = true;
  });
  let n = 1;
  return skus.map(function (s) {
    if (s._deleted || s.skuCode) return s;
    let code = styleNo + '-' + pad2(n);
    while (used[code]) {
      n += 1;
      code = styleNo + '-' + pad2(n);
    }
    used[code] = true;
    return Object.assign({}, s, { skuCode: code });
  });
}

/** 库存变更后的克重（毫克整数计算）；输入不合法返回 null */
function previewAfter(type, current, input) {
  if (input === '' || input === null || input === undefined) return null;
  const value = Number(input);
  if (!isFinite(value) || value < 0) return null;
  const cur = Math.round(Number(current) * 1000);
  const mg = Math.round(value * 1000);
  if (type === 'ADJUST') return mg / 1000;
  return (type === 'IN' ? cur + mg : cur - mg) / 1000;
}

/** 让上一页下次显示时刷新列表 */
function markPrevDirty() {
  const pages = getCurrentPages();
  const prev = pages[pages.length - 2];
  if (prev && typeof prev.markDirty === 'function') prev.markDirty();
}

module.exports = {

  fillSkuCodes: fillSkuCodes,
  previewAfter: previewAfter,
  markPrevDirty: markPrevDirty,
```

```js
/**
 * 选图、压缩到 600KB 以内并上传
 * wx.uploadFile 不走 utils/request.js，这里单独处理登录态和返回解析
 */
const { BASE_URL, TOKEN_KEY } = require('../config/index');
const { goLogin } = require('./request');

const MAX_BYTES = 600 * 1024;
const STEPS = [
  { width: 1400, quality: 80 },
  { width: 1200, quality: 70 },
  { width: 1000, quality: 60 },
  { width: 800, quality: 50 },
];
const CONCURRENCY = 3;

function choose(count) {
  return new Promise(function (resolve, reject) {
    wx.chooseMedia({
      count: count,
      mediaType: ['image'],
      sourceType: ['album', 'camera'],
      sizeType: ['compressed'],
      success: function (res) {
        resolve(res.tempFiles);
      },
      fail: function (err) {
        const msg = (err && err.errMsg) || '';
        if (msg.indexOf('cancel') > -1) resolve([]);
        else reject(new Error('选择图片失败'));
      },
    });
  });
}

function sizeOf(path) {
  return new Promise(function (resolve, reject) {
    wx.getFileSystemManager().getFileInfo({
      filePath: path,
      success: function (res) {
        resolve(res.size);
      },
      fail: function () {
        reject(new Error('读取图片失败'));
      },
    });
  });
}

function compress(path, step) {
  return new Promise(function (resolve, reject) {
    wx.compressImage({
      src: path,
      quality: step.quality,
      compressedWidth: step.width,
      success: function (res) {
        resolve(res.tempFilePath);
      },
      fail: function () {
        reject(new Error('压缩图片失败'));
      },
    });
  });
}

/** 依次尝试压缩档位，每次都从原图压缩，直到不超过 600KB */
function shrink(file) {
  if (file.size <= MAX_BYTES) return Promise.resolve(file.tempFilePath);
  function attempt(i) {
    if (i >= STEPS.length) return Promise.reject(new Error('图片太大，压缩后仍超过 600KB'));
    return compress(file.tempFilePath, STEPS[i]).then(function (path) {
      return sizeOf(path).then(function (size) {
        return size <= MAX_BYTES ? path : attempt(i + 1);
      });
    });
  }
  return attempt(0);
}

function upload(path) {
  return new Promise(function (resolve, reject) {
    wx.uploadFile({
      url: BASE_URL + '/api/mall/manage/upload',
      filePath: path,
      name: 'file',
      timeout: 30000,
      header: { Authorization: 'Bearer ' + wx.getStorageSync(TOKEN_KEY) },
      success: function (res) {
        let body = null;
        try {
          body = JSON.parse(res.data);
        } catch (e) {
          reject(new Error('上传失败（HTTP ' + res.statusCode + '）'));
          return;
        }
        if (body.code === 0) {
          resolve(body.data.url);
          return;
        }
        if (body.code === 401) goLogin();
        reject(new Error(body.msg || '上传失败'));
      },
      fail: function (err) {
        const msg = (err && err.errMsg) || '';
        if (msg.indexOf('domain') > -1) reject(new Error('上传域名未配置，请在公众平台加入 uploadFile 合法域名'));
        else reject(new Error('网络异常，上传失败'));
      },
    });
  });
}

function processOne(file) {
  return shrink(file)
    .then(upload)
    .then(function (url) {
      return { url: url };
    })
    .catch(function (e) {
      return { error: e.message || '上传失败' };
    });
}

/** 限制并发，结果顺序与入参一致 */
function runAll(files) {
  const results = new Array(files.length);
  let next = 0;
  function worker() {
    if (next >= files.length) return Promise.resolve();
    const index = next;
    next += 1;
    return processOne(files[index]).then(function (result) {
      results[index] = result;
      return worker();
    });
  }
  const workers = [];
  for (let i = 0; i < Math.min(CONCURRENCY, files.length); i += 1) workers.push(worker());
  return Promise.all(workers).then(function () {
    return results;
  });
}

/**
 * 选图并上传。resolve {urls, failed, firstError}；用户取消时 urls 为空、failed 为 0
 */
function chooseAndUpload(options) {
  return choose(options.count).then(function (files) {
    if (!files.length) return { urls: [], failed: 0, firstError: '' };
    wx.showLoading({ title: '上传中', mask: true });
    return runAll(files).then(function (results) {
      wx.hideLoading();
      const urls = results.filter(function (r) { return r.url; }).map(function (r) { return r.url; });
      const errors = results.filter(function (r) { return r.error; });
      return { urls: urls, failed: errors.length, firstError: errors.length ? errors[0].error : '' };
    });
  });
}

module.exports = { chooseAndUpload: chooseAndUpload };
```

```js
const { PAGE_SIZE } = require('../../config/index');
const { get, errMsg } = require('../../utils/request');
const { imageUrl } = require('../../utils/asset');
const { fmtWeight } = require('../../utils/format');
const { levelOf } = require('../../utils/manage');

const TABS = [
  { key: '', label: '全部' },
  { key: '1', label: '上架' },
  { key: '0', label: '下架' },
];

function decorate(row) {
  const level = levelOf(row.stockLevel);
  return Object.assign({}, row, {
    coverUrl: imageUrl(row.cover),
    availableText: fmtWeight(row.availableWeight),
    levelText: level.text,
    levelCls: level.cls,
  });
}

Page({
  data: {
    tabs: TABS,
    status: '',
    keyword: '',
    list: [],
    total: 0,
    pageNum: 1,
    noMore: false,
    loadingMore: false,
    viewState: 'loading',
    errorText: '',
  },

  onLoad: function () {
    this.seq = 0;
    this.dirty = false;
    this.reload();
  },

  onShow: function () {
    if (this.dirty) {
      this.dirty = false;
      this.reload();
    }
  },

  markDirty: function () {
    this.dirty = true;
  },

  onPullDownRefresh: function () {
    this.fetch(1).then(function () {
      wx.stopPullDownRefresh();
    });
  },

  onReachBottom: function () {
    if (this.data.noMore || this.data.loadingMore || this.data.viewState !== 'ready') return;
    this.setData({ loadingMore: true });
    this.fetch(this.data.pageNum + 1);
  },

  reload: function () {
    this.setData({ viewState: 'loading' });
    return this.fetch(1);
  },

  fetch: function (pageNum) {
    const that = this;
    this.seq += 1;
    const seq = this.seq;
    return get('/api/mall/manage/products', {
      keyword: this.data.keyword,
      status: this.data.status,
      pageNum: pageNum,
      pageSize: PAGE_SIZE,
    })
      .then(function (res) {
        if (seq !== that.seq) return;
        const rows = res.records.map(decorate);
        const list = pageNum === 1 ? rows : that.data.list.concat(rows);
        that.setData({
          list: list,
          total: res.total,
          pageNum: pageNum,
          noMore: list.length >= res.total,
          loadingMore: false,
          viewState: list.length ? 'ready' : 'empty',
        });
      })
      .catch(function (e) {
        if (seq !== that.seq) return;
        if (pageNum === 1) that.setData({ viewState: 'error', errorText: errMsg(e) });
        else {
          that.setData({ loadingMore: false });
          wx.showToast({ title: errMsg(e), icon: 'none' });
        }
      });
  },

  onTab: function (e) {
    this.setData({ status: e.currentTarget.dataset.key });
    this.reload();
  },

  onKeywordInput: function (e) {
    this.setData({ keyword: e.detail.value });
  },

  onSearch: function () {
    this.reload();
  },

  onItem: function (e) {
    wx.navigateTo({ url: '/pages/manage-product-edit/manage-product-edit?id=' + e.currentTarget.dataset.id });
  },

  onAdd: function () {
    wx.navigateTo({ url: '/pages/manage-product-edit/manage-product-edit' });
  },
});
```

```js
    name: '',

    cover: '',
    images: [],
    purity: '',
    sizeDesc: '',
    laborFeePerG: '',
    attrs: [],
    isRecommend: false,

function emptySku() {
  return { skuCode: '', specName: '', refWeight: '', refQty: '', warnWeight: '0', status: 1, initialWeight: '' };

function toText(value) {
  return value === null || value === undefined ? '' : String(value);
}

function skuFromApi(s) {
  return {
    id: s.id,
    skuCode: s.skuCode,
    specName: s.specName,
    refWeight: toText(s.refWeight),
    refQty: toText(s.refQty),
    warnWeight: toText(s.warnWeight === null || s.warnWeight === undefined ? 0 : s.warnWeight),
    status: s.status,
    initialWeight: '',
    availableText: fmtWeight(s.availableWeight),
  };
}

function numberOrNull(text) {
  return text === '' ? null : Number(text);
}

function labelled(name, status) {
  return status === 0 ? name + '（已停用）' : name;
}

Page({

    id: 0,
    viewState: 'loading',
    errorText: '',

    maxImages: MAX_IMAGES,
    categories: [],
    materials: [],
    categoryIndex: -1,
    materialIndex: -1,

          return { id: m.id, label: labelled(m.name, m.status) };

        if (that.data.id) return that.loadDetail();
        that.setData({ viewState: 'ready', skus: [emptySku()] });

        that.setData({ viewState: 'error', errorText: errMsg(e) });

  onStyleNo: function (e) {
    const value = e.detail.value.toUpperCase();

    return value;
  },

  onCategory: function (e) {
    this.setData({ categoryIndex: Number(e.detail.value) });
  },

    this.setData({ materialIndex: Number(e.detail.value) });

    this.setData({ 'form.isRecommend': e.detail.value });

  reportUpload: function (result) {
    if (result.failed) {
      wx.showToast({ title: result.failed + ' 张失败：' + result.firstError, icon: 'none', duration: 3000 });
    }
  },

  pickCover: function () {
    const that = this;
    chooseAndUpload({ count: 1 })
      .then(function (result) {
        that.reportUpload(result);
        if (!result.urls.length) return;
        that.setData({ 'form.cover': result.urls[0] });
        that.syncImages();
      })
      .catch(function (e) {
        wx.showToast({ title: errMsg(e), icon: 'none' });
      });
  },

  onCover: function () {
    const that = this;
    if (!this.data.form.cover) {
      this.pickCover();
      return;
    }
    wx.showActionSheet({
      itemList: ['更换', '删除'],
      success: function (r) {
        if (r.tapIndex === 0) that.pickCover();
        else {
          that.setData({ 'form.cover': '' });
          that.syncImages();
        }
      },
    });
  },

  onAddImages: function () {
    const that = this;
    const left = MAX_IMAGES - this.data.form.images.length;
    if (left <= 0) return;
    chooseAndUpload({ count: left })
      .then(function (result) {
        that.reportUpload(result);
        if (!result.urls.length) return;
        that.setData({ 'form.images': that.data.form.images.concat(result.urls).slice(0, MAX_IMAGES) });
        that.syncImages();
      })
      .catch(function (e) {
        wx.showToast({ title: errMsg(e), icon: 'none' });
      });
  },

  onImage: function (e) {
    const that = this;
    const index = e.currentTarget.dataset.index;
    wx.showActionSheet({
      itemList: ['设为主图', '删除'],
      success: function (r) {
        const images = that.data.form.images.slice();
        const picked = images[index];
        if (r.tapIndex === 0) {
          const oldCover = that.data.form.cover;
          if (oldCover) images[index] = oldCover;
          else images.splice(index, 1);
          that.setData({ 'form.cover': picked, 'form.images': images });
        } else {
          images.splice(index, 1);
          that.setData({ 'form.images': images });
        }
        that.syncImages();
      },
    });
  },

  // ── 规格 ──
  onAddSku: function () {
    this.setData({ sheetShow: true, skuIndex: -1, skuForm: emptySku() });
  },

  onEditSku: function (e) {
    const index = e.currentTarget.dataset.index;
    this.setData({ sheetShow: true, skuIndex: index, skuForm: Object.assign({}, this.data.skus[index]) });
  },

  closeSheet: function () {
    this.setData({ sheetShow: false });
  },

  noop: function () {},

  onSkuInput: function (e) {
    this.setData({ ['skuForm.' + e.currentTarget.dataset.field]: e.detail.value });
  },

  onSkuStatus: function (e) {
    this.setData({ 'skuForm.status': e.detail.value ? 1 : 0 });
  },

  saveSku: function () {
    const sku = this.data.skuForm;
    if (!sku.specName.trim()) {
      wx.showToast({ title: '请填写规格名', icon: 'none' });
      return;
    }
    const skus = this.data.skus.slice();
    if (this.data.skuIndex >= 0) skus[this.data.skuIndex] = sku;
    else skus.push(sku);
    this.setData({ skus: skus, sheetShow: false });
  },

  removeSku: function () {
    const that = this;
    const index = this.data.skuIndex;
    if (index < 0) {
      this.setData({ sheetShow: false });
      return;
    }
    const sku = this.data.skus[index];
    function apply() {
      const skus = that.data.skus.slice();
      if (sku.id) skus[index] = Object.assign({}, sku, { _deleted: true });
      else skus.splice(index, 1);
      that.setData({ skus: skus, sheetShow: false });
    }
    if (!sku.id) {
      apply();
      return;
    }
    wx.showModal({
      title: '删除规格',
      content: '确定删除「' + sku.specName + '」吗？库存必须为 0，点保存后生效。',
      success: function (r) {
        if (r.confirm) apply();
      },
    });
  },

  // ── 自定义属性 ──
  onAddAttr: function () {
    this.setData({ 'form.attrs': this.data.form.attrs.concat([{ name: '', value: '' }]) });
  },

  onAttrInput: function (e) {
    const d = e.currentTarget.dataset;
    this.setData({ ['form.attrs[' + d.index + '].' + d.key]: e.detail.value });
  },

  onRemoveAttr: function (e) {
    const attrs = this.data.form.attrs.slice();
    attrs.splice(e.currentTarget.dataset.index, 1);
    this.setData({ 'form.attrs': attrs });
  },

  // ── 保存与操作 ──
  buildPayload: function () {
    const f = this.data.form;
    const category = this.data.categories[this.data.categoryIndex];
    const material = this.data.materials[this.data.materialIndex];
    const styleNo = f.styleNo.trim();
    const skus = fillSkuCodes(this.data.skus, styleNo);
    // 不提交 detailHtml 和 status：图文详情保留 PC 内容，上下架走单独接口
    return {
      name: f.name.trim(),
      styleNo: styleNo,
      categoryId: category ? category.id : 0,
      materialId: material ? material.id : 0,
      cover: f.cover,
      images: f.images,
      purity: f.purity,
      sizeDesc: f.sizeDesc,
      laborFeePerG: f.laborFeePerG,
      attrs: f.attrs,
      isRecommend: f.isRecommend,
      sort: f.sort === '' ? 100 : Number(f.sort),
      skus: skus.map(function (s) {
        return {
          id: s.id,
          skuCode: s.skuCode,
          specName: s.specName,
          refWeight: numberOrNull(s.refWeight),
          refQty: numberOrNull(s.refQty),
          warnWeight: s.warnWeight === '' ? 0 : Number(s.warnWeight),
          status: s.status,
          initialWeight: s.id || s.initialWeight === '' ? undefined : Number(s.initialWeight),
          _deleted: s._deleted,
        };
      }),
    };
  },

  /** 校验款号后保存；新建返回新 id，编辑返回当前 id */
  persist: function () {
    if (!this.data.form.styleNo.trim()) return Promise.reject(new Error('请填写款式编号'));
    const payload = this.buildPayload();
    if (this.data.id) {
      const id = this.data.id;
      return put('/api/mall/manage/products/' + id, payload).then(function () {
        return id;
      });
    }
    return post('/api/mall/manage/products', payload).then(function (res) {
      return res.id;
    });
  },

  save: function () {
    const that = this;
    if (this.data.saving) return;
    this.setData({ saving: true });
    const isNew = !this.data.id;
    this.persist()
      .then(function (id) {
        markPrevDirty();
        wx.showToast({ title: '已保存', icon: 'success' });
        if (isNew) {
          wx.redirectTo({ url: '/pages/manage-product-edit/manage-product-edit?id=' + id });
          return;
        }
        return that.loadDetail();
      })
      .catch(function (e) {
        wx.showToast({ title: errMsg(e), icon: 'none', duration: 3000 });
      })
      .then(function () {
        that.setData({ saving: false });
      });
  },

  /** 先保存表单再切换上下架，避免按旧数据校验 */
  onToggleStatus: function () {
    const that = this;
    if (this.data.busy) return;
    const next = this.data.form.status === 1 ? 0 : 1;
    this.setData({ busy: true });
    this.persist()
      .then(function (id) {
        return put('/api/mall/manage/products/' + id + '/status', { status: next });
      })
      .then(function () {
        markPrevDirty();
        wx.showToast({ title: next === 1 ? '已上架' : '已下架', icon: 'success' });
        return that.loadDetail();
      })
      .catch(function (e) {
        wx.showToast({ title: errMsg(e), icon: 'none', duration: 3000 });
      })
      .then(function () {
        that.setData({ busy: false });
      });
  },

  onCopy: function () {
    const that = this;
    if (this.data.busy) return;
    this.setData({ busy: true });
    post('/api/mall/manage/products/' + this.data.id + '/copy')
      .then(function (res) {
        markPrevDirty();
        wx.showToast({ title: '已复制', icon: 'success' });
        wx.redirectTo({ url: '/pages/manage-product-edit/manage-product-edit?id=' + res.id });
      })
      .catch(function (e) {
        wx.showToast({ title: errMsg(e), icon: 'none', duration: 3000 });
      })
      .then(function () {
        that.setData({ busy: false });
      });
  },

  onDelete: function () {
    const that = this;
    wx.showModal({
      title: '删除商品',
      content: '删除后前台不再显示，确定删除吗？',
      confirmColor: '#dc2626',
      success: function (r) {
        if (!r.confirm) return;
        request({ url: '/api/mall/manage/products/' + that.data.id, method: 'DELETE' })
          .then(function () {
            markPrevDirty();
            wx.showToast({ title: '已删除', icon: 'success' });
            setTimeout(function () {
              wx.navigateBack();
            }, 600);
          })
          .catch(function (e) {
            wx.showToast({ title: errMsg(e), icon: 'none', duration: 3000 });
          });
      },
    });
  },
});
```

```js
const { PAGE_SIZE } = require('../../config/index');
const { get, post, errMsg } = require('../../utils/request');
const { fmtWeight } = require('../../utils/format');
const { levelOf, previewAfter } = require('../../utils/manage');

const OPS = [
  { key: 'IN', path: 'in', label: '入库', inputLabel: '入库克重 (g)', reasons: ['新货入库', '客户退货', '盘盈补录'] },
  { key: 'OUT', path: 'out', label: '出库', inputLabel: '出库克重 (g)', reasons: ['线下售出', '调拨出库', '损耗'] },
  { key: 'ADJUST', path: 'adjust', label: '盘点', inputLabel: '盘点后总克重 (g)', reasons: ['定期盘点', '称重修正'] },
];

function decorate(row) {
  const level = levelOf(row.level);
  return Object.assign({}, row, {
    stockText: fmtWeight(row.stockWeight),
    lockedText: fmtWeight(row.lockedWeight),
    availableText: fmtWeight(row.availableWeight),
    levelText: level.text,
    levelCls: level.cls,
  });
}

function emptyPanel() {
  return { show: false, sku: null, opIndex: 0, weight: '', reason: '', refNo: '', previewText: '', previewBad: false };
}

Page({
  data: {
    ops: OPS,
    keyword: '',
    onlyWarn: false,
    summary: null,
    summaryView: null,
    list: [],
    total: 0,
    pageNum: 1,
    noMore: false,
    loadingMore: false,
    viewState: 'loading',
    errorText: '',
    panel: emptyPanel(),
    submitting: false,
  },

  onLoad: function () {
    this.seq = 0;
    this.reload();
  },

  onPullDownRefresh: function () {
    this.fetch(1).then(function () {
      wx.stopPullDownRefresh();
    });
  },

  onReachBottom: function () {
    if (this.data.noMore || this.data.loadingMore || this.data.viewState !== 'ready') return;
    this.setData({ loadingMore: true });
    this.fetch(this.data.pageNum + 1);
  },

  reload: function () {
    this.setData({ viewState: 'loading' });
    return this.fetch(1);
  },

  fetch: function (pageNum) {
    const that = this;
    this.seq += 1;
    const seq = this.seq;
    return get('/api/mall/manage/stocks', {
      keyword: this.data.keyword,
      onlyWarn: this.data.onlyWarn ? '1' : '',
      pageNum: pageNum,
      pageSize: PAGE_SIZE,
    })
      .then(function (res) {
        if (seq !== that.seq) return;
        const rows = res.records.map(decorate);
        const list = pageNum === 1 ? rows : that.data.list.concat(rows);
        that.setData({
          list: list,
          total: res.total,
          pageNum: pageNum,
          noMore: list.length >= res.total,
          loadingMore: false,
          viewState: list.length ? 'ready' : 'empty',
          summaryView: {
            skuCount: res.summary.skuCount,
            warnCount: res.summary.warnCount,
            stock: fmtWeight(res.summary.stockWeight),
            locked: fmtWeight(res.summary.lockedWeight),
            available: fmtWeight(res.summary.availableWeight),
          },
        });
      })
      .catch(function (e) {
        if (seq !== that.seq) return;
        if (pageNum === 1) that.setData({ viewState: 'error', errorText: errMsg(e) });
        else {
          that.setData({ loadingMore: false });
          wx.showToast({ title: errMsg(e), icon: 'none' });
        }
      });
  },

  onKeywordInput: function (e) {
    this.setData({ keyword: e.detail.value });
  },

  onSearch: function () {
    this.reload();
  },

  onWarn: function (e) {
    this.setData({ onlyWarn: e.detail.value });
    this.reload();
  },

  goLogs: function () {
    wx.navigateTo({ url: '/pages/manage-stock-logs/manage-stock-logs' });
  },

  goSkuLogs: function () {
    const sku = this.data.panel.sku;
    this.closePanel();
    wx.navigateTo({
      url:
        '/pages/manage-stock-logs/manage-stock-logs?skuId=' +
        sku.id +
        '&name=' +
        encodeURIComponent(sku.productName + ' ' + sku.specName),
    });
  },

  // ── 操作面板 ──
  onItem: function (e) {
    const sku = this.data.list[e.currentTarget.dataset.index];
    this.setData({ panel: Object.assign(emptyPanel(), { show: true, sku: sku }) });
  },

  closePanel: function () {
    this.setData({ 'panel.show': false });
  },

  noop: function () {},

  refreshPreview: function () {
    const panel = this.data.panel;
    const after = previewAfter(OPS[panel.opIndex].key, panel.sku.stockWeight, panel.weight);
    if (after === null) {
      this.setData({ 'panel.previewText': '', 'panel.previewBad': false });
      return;
    }
    const bad = after < panel.sku.lockedWeight;
    this.setData({
      'panel.previewText': bad ? '变更后 ' + after + ' g，低于已预占 ' + panel.sku.lockedText + ' g' : '变更后 ' + fmtWeight(after) + ' g',
      'panel.previewBad': bad,
    });
  },

  onOp: function (e) {
    this.setData({ 'panel.opIndex': e.currentTarget.dataset.index, 'panel.reason': '' });
    this.refreshPreview();
  },

  onWeight: function (e) {
    this.setData({ 'panel.weight': e.detail.value });
    this.refreshPreview();
  },

  onReason: function (e) {
    this.setData({ 'panel.reason': e.detail.value });
  },

  onReasonChip: function (e) {
    this.setData({ 'panel.reason': e.currentTarget.dataset.value });
  },

  onRefNo: function (e) {
    this.setData({ 'panel.refNo': e.detail.value });
  },

  onSubmit: function () {
    const that = this;
    const panel = this.data.panel;
    const op = OPS[panel.opIndex];
    if (panel.weight === '' || !isFinite(Number(panel.weight))) {
      wx.showToast({ title: '请填写克重', icon: 'none' });
      return;
    }
    if (!panel.reason.trim()) {
      wx.showToast({ title: '请填写原因', icon: 'none' });
      return;
    }
    if (op.key === 'IN') {
      this.submit(op);
      return;
    }
    wx.showModal({
      title: '确认' + op.label,
      content: panel.sku.productName + ' ' + panel.sku.specName + '，' + op.inputLabel + '：' + panel.weight,
      success: function (r) {
        if (r.confirm) that.submit(op);
      },
    });
  },

  submit: function (op) {
    const that = this;
    if (this.data.submitting) return;
    const panel = this.data.panel;
    this.setData({ submitting: true });
    post('/api/mall/manage/stocks/' + op.path, {
      skuId: panel.sku.id,
      weight: Number(panel.weight),
      reason: panel.reason.trim(),
      refNo: panel.refNo.trim(),
    })
      .then(function () {
        wx.showToast({ title: op.label + '成功', icon: 'success' });
        that.setData({ 'panel.show': false });
        return that.fetch(1);
      })
      .catch(function (e) {
        wx.showToast({ title: errMsg(e), icon: 'none', duration: 3000 });
      })
      .then(function () {
        that.setData({ submitting: false });
      });
  },
});
```

```js
const { PAGE_SIZE } = require('../../config/index');
const { get, errMsg } = require('../../utils/request');
const { fmtDateTime, fmtWeight } = require('../../utils/format');

const TYPES = [
  { key: '', label: '全部' },
  { key: 'IN', label: '入库' },
  { key: 'OUT', label: '出库' },
  { key: 'ADJUST', label: '盘点调整' },
  { key: 'ORDER_OUT', label: '下单扣减' },
  { key: 'ORDER_BACK', label: '取消回库' },
];

const TYPE_LABEL = {
  IN: '入库',
  OUT: '出库',
  ADJUST: '盘点调整',
  LOCK: '下单预占',
  WEIGH_OUT: '称重出库',
  UNLOCK: '取消释放',
  REFUND_BACK: '退款回库',
  ORDER_OUT: '下单扣减',
  ORDER_BACK: '取消回库',
};

function decorate(log) {
  return Object.assign({}, log, {
    typeLabel: TYPE_LABEL[log.type] || log.type,
    changeText: (log.changeWeight > 0 ? '+' : '') + fmtWeight(log.changeWeight),
    changeCls: log.changeWeight > 0 ? 'plus' : 'minus',
    beforeText: fmtWeight(log.beforeWeight),
    afterText: fmtWeight(log.afterWeight),
    timeText: fmtDateTime(log.createdAt),
  });
}

Page({
  data: {
    types: TYPES,
    type: '',
    keyword: '',
    startDate: '',
    endDate: '',
    skuId: 0,
    skuName: '',
    list: [],
    pageNum: 1,
    noMore: false,
    loadingMore: false,
    viewState: 'loading',
    errorText: '',
  },

  onLoad: function (options) {
    this.seq = 0;
    this.setData({
      skuId: Number(options.skuId || 0),
      skuName: options.name ? decodeURIComponent(options.name) : '',
    });
    this.reload();
  },

  onPullDownRefresh: function () {
    this.fetch(1).then(function () {
      wx.stopPullDownRefresh();
    });
  },

  onReachBottom: function () {
    if (this.data.noMore || this.data.loadingMore || this.data.viewState !== 'ready') return;
    this.setData({ loadingMore: true });
    this.fetch(this.data.pageNum + 1);
  },

  reload: function () {
    this.setData({ viewState: 'loading' });
    return this.fetch(1);
  },

  fetch: function (pageNum) {
    const that = this;
    this.seq += 1;
    const seq = this.seq;
    return get('/api/mall/manage/stock-logs', {
      keyword: this.data.keyword,
      type: this.data.type,
      skuId: this.data.skuId || '',
      startDate: this.data.startDate,
      endDate: this.data.endDate,
      pageNum: pageNum,
      pageSize: PAGE_SIZE,
    })
      .then(function (res) {
        if (seq !== that.seq) return;
        const rows = res.records.map(decorate);
        const list = pageNum === 1 ? rows : that.data.list.concat(rows);
        that.setData({
          list: list,
          pageNum: pageNum,
          noMore: list.length >= res.total,
          loadingMore: false,
          viewState: list.length ? 'ready' : 'empty',
        });
      })
      .catch(function (e) {
        if (seq !== that.seq) return;
        if (pageNum === 1) that.setData({ viewState: 'error', errorText: errMsg(e) });
        else {
          that.setData({ loadingMore: false });
          wx.showToast({ title: errMsg(e), icon: 'none' });
        }
      });
  },

  onType: function (e) {
    this.setData({ type: e.currentTarget.dataset.key });
    this.reload();
  },

  onKeywordInput: function (e) {
    this.setData({ keyword: e.detail.value });
  },

  onSearch: function () {
    this.reload();
  },

  onStart: function (e) {
    this.setData({ startDate: e.detail.value });
    this.reload();
  },

  onEnd: function (e) {
    this.setData({ endDate: e.detail.value });
    this.reload();
  },

  clearDates: function () {
    this.setData({ startDate: '', endDate: '' });
    this.reload();
  },

  clearSku: function () {
    this.setData({ skuId: 0, skuName: '' });
    this.reload();
  },
});
```
